Provides the ``ProjectManager`` class for searching, creating, locking,
and unlocking analytical projects.

_USE_INSTANCE_DOMAIN = object()
"""Sentinel indicating that ``list_projects`` should use the instance domain."""


def _reduce_umbrella_name_elements(
    elements: list[RuleSetUmbrellaNameElement | None] | None,
) -> tuple[NameValuePair, ...]:
    """Reduce BCMS umbrella name elements to ``NameValuePair`` entries.

    ``None`` entries are skipped and each element's ``value`` defaults to
    an empty string when it is absent or null (``value`` is a required key
    of :class:`~statefarm.bcontent.sdk.api.bcms.NameValuePair`).

    Parameters
    ----------
    elements : list[RuleSetUmbrellaNameElement | None] | None
        The umbrella ``nameElements`` payload from BCMS.

    Returns
    -------
    tuple[NameValuePair, ...]
        The reduced name/value pairs.
    """
    pairs: list[NameValuePair] = []
    for element in elements or []:
        if element is None:
            continue
        pairs.append({"name": element["name"], "value": element.get("value") or ""})
    return tuple(pairs)


class ProjectManager:
    """Manage analytical projects via BCMS.

    ``ProjectManager`` is the main entry-point for discovering, creating,
    and administering analytical projects stored in BCMS.

    Parameters
    ----------
    client : BCMSAPIClient
        An authenticated BCMS API client instance.
    domain : str | None, optional
        Domain to scope operations to.  Must be a valid domain name
        known to BCMS.  When *None* (the default), operations are not
        scoped to any single domain.

    Raises
    ------
    ValueError
        If *domain* is provided but is not a recognised BCMS domain.

    Examples
    --------
    >>> from statefarm.bcontent.sdk.api import BCMSAPIClient
    >>> client = BCMSAPIClient(...)
    >>> pm = ProjectManager(client, domain="auto_rating")
    >>> pm.list_projects()
    [ProjectSummary(identifier=..., name='my_project', businessName=..., comment=..., domain=..., nameElements=(...))]
    """

    def __init__(
        self,
        client: BCMSAPIClient,
        domain: str | None = None,
    ) -> None:
        self._client = client

        domains = client.query.rule_set_domains(limit=1000)
        self._domain_allowed: list[str] = [d["name"] for d in (domains["items"] or [])]

        # Initialize _domain before using setter
        self._domain: str | None = None
        # Use the setter for validation
        self.domain = domain

        self._classification: str = "analytical_project"

    # ------------------------------------------------------------------
    # Properties
    # ------------------------------------------------------------------

    @property
    def client(self) -> BCMSAPIClient:
        """Get the BCMS API client.

        Returns
        -------
        BCMSAPIClient
            The client instance provided at construction time.
        """
        return self._client

    @property
    def domain(self) -> str | None:
        """Get or set the active domain.

        Returns
        -------
        str | None
            The currently active domain, or *None* if unset.
        """
        return self._domain

    @domain.setter
    def domain(self, value: str | None) -> None:
        """Set the active domain, validating against allowed domains.

        Parameters
        ----------
        value : str | None
            Domain name to set.  Pass *None* to clear the domain.

        Raises
        ------
        ValueError
            If *value* is not a recognised BCMS domain.
        """
        if value is None:
            self._domain = None
        else:
            if value not in self._domain_allowed:
                raise ValueError(
                    f"Invalid domain '{value}'. "
                    f"Allowed domains are: {', '.join(self._domain_allowed)}"
                )
            self._domain = value

    @property
    def classification(self) -> str:
        """Get the project classification.

        Returns
        -------
        str
            The classification string used for filtering projects.
        """
        return self._classification

    # ------------------------------------------------------------------
    # Public methods
    # ------------------------------------------------------------------

    def _resolve_effective_domain(
        self, domain_override: str | None | object
    ) -> str | None:
        """Resolve the effective domain for a query.

        Parameters
        ----------
        domain_override : str | None | object
            The domain override value from the caller.  The sentinel
            ``_USE_INSTANCE_DOMAIN`` means "use the instance domain".

        Returns
        -------
        str | None
            The resolved domain name, or *None* for all domains.

        Raises
        ------
        ValueError
            If *domain_override* is a string not in the allowed list.
        """
        if domain_override is _USE_INSTANCE_DOMAIN:
            return self._domain

        if isinstance(domain_override, str):
            if domain_override not in self._domain_allowed:
                raise ValueError(
                    f"Invalid domain '{domain_override}'. "
                    f"Allowed domains are: "
                    f"{', '.join(self._domain_allowed)}"
                )
            return domain_override
        return None

    def list_projects(
        self,
        project_name: str | None = None,
        name_elements: list[NameValuePair] | None = None,
        *,
        domain_override: str | None | object = _USE_INSTANCE_DOMAIN,
        classification: str | None = None,
    ) -> list[ProjectSummary]:
        """List analytical projects, optionally filtered by name or domain.

        Queries BCMS for umbrellas matching the analytical-project
        classification.  When *project_name* is given the
        ``businessNames`` server-side filter is applied so only matching
        records are returned by BCMS.

        Parameters
        ----------
        project_name : str | None, optional
            If provided, only projects whose ``businessName`` matches
            exactly are returned.
        name_elements : list[NameValuePair] | None, optional
            If provided, only projects with matching name elements are
            returned.  Cannot be combined with *project_name*.
        domain_override : str | None, optional, keyword-only
            Override the instance domain filter for this query.

            * Not provided (default) — uses the instance's ``domain``
              setting.
            * A string — queries only that domain (must be a valid
              BCMS domain name).
            * ``None`` — queries across **all** domains regardless of
              the instance setting.
        classification : str | None, optional, keyword-only
            Classification to filter by.  Defaults to the instance
            classification.

        Returns
        -------
        list[ProjectSummary]
            Each :class:`ProjectSummary` exposes ``identifier``, ``name``,
            ``businessName``, ``comment``, ``domain``, and
            ``nameElements`` as attributes.

        Raises
        ------
        ValueError
            If both *project_name* and *name_elements* are provided,
            or if *domain_override* is a string that is not in the
            allowed domains list.

        Notes
        -----
        The *domain_override* parameter does not modify the instance's
        ``domain`` property.  It only affects the current query.

        The *project_name* and *name_elements* parameters are mutually
        exclusive.  Provide at most one of them.
        """
        if project_name is not None and name_elements is not None:
            raise ValueError(
                "Cannot specify both 'project_name' and 'name_elements'. "
                "Provide only one filtering option."
            )
        filter_input: RuleSetUmbrellaFilterInput = {
            "classifications": {"name": classification or self._classification},
        }

        effective_domain = self._resolve_effective_domain(domain_override)

        if effective_domain is not None:
            filter_input["domains"] = {"name": effective_domain}
        if project_name is not None:
            filter_input["businessNames"] = project_name
        if name_elements is not None:
            filter_input["nameElements"] = [
                RuleSetUmbrellaFilterNameElementInput(
                    name=ne["name"], value=ne.get("value")
                )
                for ne in name_elements
            ]

        projects: list[ProjectSummary] = []
        for umbrella in _query_umbrellas(self._client, filter_input):
            domain = umbrella["domain"]
            projects.append(
                ProjectSummary(
                    identifier=umbrella["identifier"],
                    name=umbrella["alternateName"],
                    businessName=umbrella["businessName"],
                    comment=umbrella["comment"],
                    domain=domain["name"] if domain else "",
                    nameElements=_reduce_umbrella_name_elements(
                        umbrella.get("nameElements")
                    ),
                )
            )
        return projects

    def _get_next_version(self, name_elements: list[NameValuePair]) -> str:
        """Determine the next version for a project with matching name elements.

        Queries BCMS for existing projects whose name elements match
        *name_elements* (ignoring the ``version`` element).  The query is
        scoped to neither classification nor domain, because project names
        are unique across all of them.  Returns the next sequential version
        as a 7-digit zero-padded string.

        Parameters
        ----------
        name_elements : list[NameValuePair]
            Name elements (without version) to match against.

        Returns
        -------
        str
            The next version string (e.g., ``'0001001'``, ``'0001002'``).
        """
        filter_input: RuleSetUmbrellaFilterInput = {
            "nameElements": [
                RuleSetUmbrellaFilterNameElementInput(
                    name=ne["name"], value=ne.get("value")
                )
                for ne in name_elements
            ]
        }

        versions: list[int] = []
        for umbrella in _query_umbrellas(self._client, filter_input):
            for e in _reduce_umbrella_name_elements(umbrella.get("nameElements")):
                if e["name"] == "version" and str(e["value"]).isdigit():
                    versions.append(int(e["value"]))
        return f"{max(versions, default=1000) + 1:07d}"

    def _resolve_create_domain(self, domain: str | None) -> str:
        """Resolve the domain to use when creating a project.

        Parameters
        ----------
        domain : str | None
            The domain passed to create_project.

        Returns
        -------
        str
            The resolved domain name.
        """
        if self._domain is not None and domain is not None:
            raise ValueError(
                "Cannot specify domain when ProjectManager already has a domain set."
            )

        if self._domain is not None:
            return self._domain
        if domain is not None:
            if domain not in self._domain_allowed:
                raise ValueError(
                    f"Invalid domain '{domain}'. "
                    f"Allowed domains are: "
                    f"{', '.join(self._domain_allowed)}"
                )
            return domain
        raise ValueError(
            "A domain must be provided either on the "
            "ProjectManager instance or as a parameter."
        )

    def _create_umbrella(
        self,
        classification: str,
        name_elements: list[NameValuePair],
        alt_name: str | None = None,
        domain: str | None = None,
        comment: str | None = None,
        properties: list[NameValuePair] | None = None,
    ) -> UUID:
        """Create an umbrella under *classification* and return its identifier."""
        domain_value = self._resolve_create_domain(domain)

        for elem in name_elements:
            if elem.get("name") == "version":
                raise ValueError(
                    "Cannot specify 'version' in name_elements. "
                    "Version is auto-generated."
                )

        next_version = self._get_next_version(name_elements)
        elements_with_version: list[NameValueInput] = []
        for elem in name_elements:
            name, value = elem["name"], elem["value"]
            if name is not None and value is not None:
                elements_with_version.append({"name": name, "value": value})
        elements_with_version.append({"name": "version", "value": next_version})

        umbrella_input = InsertRuleSetUmbrellaInput(
            classification={"name": classification},
            domain={"name": domain_value},
        )
        if alt_name is not None:
            umbrella_input["alternateName"] = alt_name
        umbrella_input["nameElements"] = elements_with_version
        if comment is not None:
            umbrella_input["comment"] = comment
        if properties is not None:
            umbrella_input["properties"] = properties

        umbrella = self._client.mutation.insert_rule_set_umbrella(umbrella_input)
        return str(umbrella["identifier"])

    def create_project(
        self,
        name_elements: list[NameValuePair],
        alt_name: str | None = None,
        domain: str | None = None,
        comment: str | None = None,
        data_contract_name: str | None = None,
        data_contract_version: str | None = None,
    ) -> Project:
        """Create a new analytical project in BCMS.

        Version is auto-generated as a 7-digit zero-padded string
        (starting at ``'0001001'``) and appended to *name_elements*.
        Callers must **not** include a ``version`` element.

        Parameters
        ----------
        name_elements : list[NameValuePair]
            A list of name/value pairs to set on the project.  Each pair
            should have ``name`` and ``value`` keys.  Must **not** contain
            an element with ``name == "version"`` (version is
            auto-generated).
        alt_name : str | None, optional
            The alternate name for the new project.
        domain : str | None, optional
            Domain to assign the project to.  Must not be provided when
            the ``ProjectManager`` already has a domain set.
        comment : str | None, optional
            An optional comment describing the project.
        data_contract_name : str | None, optional
            Name of the data contract to associate with the project.
            When *None*, the data contract name is not set on the
            project.
        data_contract_version : str | None, optional
            Version of the data contract to associate with the project.
            When *None*, the data contract version is not set on the
            project.

        Returns
        -------
        Project
            A ``Project`` instance bound to the newly created project.

        Raises
        ------
        ValueError
            If both the instance domain and *domain* parameter are set,
            if no domain can be determined, if the resolved domain is
            not in the allowed list, or if *name_elements* contains
            a ``version`` element.
        """
        identifier = self._create_umbrella(
            self._classification,
            name_elements,
            alt_name=alt_name,
            domain=domain,
            comment=comment,
        )
        project = Project(self._client, umbrella_identifier=identifier)

        if data_contract_name is not None or data_contract_version is not None:
            project.update_data_contract(data_contract_name, data_contract_version)

        return project

    def lock_project(self, identifier: UUID) -> None:
        """Lock a project to prevent modifications.

        Parameters
        ----------
        identifier : UUID
            The BCMS umbrella identifier of the project to lock.

        Warnings
        --------
        If the project is already locked a warning is issued (including
        the project name and existing lock reason) and the method returns
        without taking any action.
        """
        umbrella = self._client.query.rule_set_umbrella(identifier=identifier)
        if umbrella is None:
            raise ValueError(f"Project with identifier '{identifier}' not found.")
        project_name = umbrella["alternateName"]

        lock_reason = umbrella["lockReason"]
        if lock_reason is not None:
            lock_reason_name = lock_reason["name"]
            warnings.warn(
                f"Project '{project_name}' ({identifier}) is already "
                f"locked with reason: {lock_reason_name}",
                stacklevel=2,
            )
            return

        self._client.mutation.lock_rule_set_umbrella(
            input=LockComponentInput(
                identifier=identifier,
                lockReason=LockReason(name="Business Lock"),
            )
        )

    def unlock_project(self, identifier: UUID) -> None:
        """Unlock a project to allow modifications.

        Only projects locked with reason "Business Lock"
        (case-insensitive) can be unlocked by ``ProjectManager``.

        Parameters
        ----------
        identifier : UUID
            The BCMS umbrella identifier of the project to unlock.

        Raises
        ------
        ValueError
            If the project is locked with a reason other than
            "Business Lock".  The error message includes the project
            name, identifier, lock reason, and responsible party
            (if available).

        Warnings
        --------
        If the project is already unlocked a warning is issued
        (including the project name) and the method returns without
        taking any action.
        """
        umbrella = self._client.query.rule_set_umbrella(identifier=identifier)
        if umbrella is None:
            raise ValueError(f"Project with identifier '{identifier}' not found.")
        project_name = umbrella["alternateName"]
        lock_reason = umbrella["lockReason"]

        if lock_reason is None:
            warnings.warn(
                f"Project '{project_name}' ({identifier}) is already unlocked.",
                stacklevel=2,
            )
            return

        lock_reason_name = lock_reason["name"]
        if lock_reason_name.lower() != "business lock":
            lock_party = umbrella.get("lockResponsibleParty")
            party_msg = f" by {lock_party}" if lock_party else ""
            raise ValueError(
                f"Cannot unlock project '{project_name}' ({identifier}). "
                f"Lock reason is '{lock_reason_name}'{party_msg}, but "
                f"only 'Business Lock' locks can be unlocked by "
                f"ProjectManager."
            )

        self._client.mutation.unlock_rule_set_umbrella(identifier=identifier)

    def get_project(
        self,
        identifier: UUID | None = None,
        project_name: str | None = None,
    ) -> Project:
        """Get a ``Project`` instance for the given umbrella.

        Exactly one of *identifier* or *project_name* must be provided.

        Parameters
        ----------
        identifier : UUID | None, optional
            The BCMS umbrella identifier.
        project_name : str | None, optional
            The business name of the project.  The method queries BCMS
            across all domains to resolve the identifier.

        Returns
        -------
        Project
            A ``Project`` bound to the specified umbrella.

        Raises
        ------
        ValueError
            If neither or both parameters are provided, if no project
            matches *project_name*, or if multiple projects match.
        """
        if identifier is None and project_name is None:
            raise ValueError("Must provide either 'identifier' or 'project_name'.")
        if identifier is not None and project_name is not None:
            raise ValueError(
                "Cannot specify both 'identifier' and 'project_name'. Provide only one."
            )

        if project_name is not None:
            projects = self.list_projects(
                project_name=project_name, domain_override=None
            )
            if len(projects) == 0:
                raise ValueError(
                    f"No project found with business name '{project_name}'."
                )
            if len(projects) > 1:
                ids = [p.identifier for p in projects]
                raise ValueError(
                    f"Multiple projects found with business name "
                    f"'{project_name}': {ids}. "
                    f"Use 'identifier' to specify which project."
                )
            identifier = projects[0].identifier

        return Project(self._client, umbrella_identifier=identifier)

    # ------------------------------------------------------------------
    # Dunder methods
    # ------------------------------------------------------------------

    def __str__(self) -> str:
        """Return a human-readable string representation."""
        return f"ProjectManager(domain={self._domain})"

    def __repr__(self) -> str:
        """Return a detailed string representation."""
        return f"ProjectManager(domain={self._domain})"

class Project:
    """Manage a single analytical project backed by BCMS.

    The Project class provides methods to manage groupings and steps
    within an analytical project stored in BCMS (Business Content
    Management System).

    Examples
    --------
    >>> project = Project(client, umbrella_identifier="umbrella-id-123")  # doctest: +SKIP
    >>> project.list_groupings()  # doctest: +SKIP
    ['model_prep', 'model']

    >>> project.create_grouping("post_processing")  # doctest: +SKIP

    >>> step = {"name": "normalize", "executionRule": "always"}  # doctest: +SKIP
    >>> project.create_step("model_prep", step)  # doctest: +SKIP
    ['normalize']

    >>> step_with_dt = {"name": "score", "executionRule": "conditional"}  # doctest: +SKIP
    >>> dt = {"name": "score_table", "data": [...], "definition": {...}}  # doctest: +SKIP
    >>> project.create_step("model", step_with_dt, decision_tables=dt)  # doctest: +SKIP
    ['score']

    >>> project.steps  # raw BCMS steps list  # doctest: +SKIP
    [{'identifier': '...', 'name': 'model_prep', ...}, ...]

    >>> project.stepMapping  # steps organized by grouping  # doctest: +SKIP
    {'model_prep': [{'name': 'normalize', ...}], 'model': [...]}

    >>> project.decisionTables  # raw BCMS decision tables list  # doctest: +SKIP
    [{'identifier': '...', 'name': 'score_table', 'data': [...], ...}, ...]

    >>> project.decisionTableMapping  # DTs by name  # doctest: +SKIP
    {'score_table': {'identifier': '...', 'name': 'score_table', ...}, ...}

    >>> project.get_steps("normalize")  # doctest: +SKIP
    {'steps': [...], 'decisionTables': [...]}

    >>> project.get_steps(groupings="model_prep")  # doctest: +SKIP
    {'steps': [...], 'decisionTables': [...]}

    >>> project.list_steps("model_prep")  # doctest: +SKIP
    ['normalize']

    >>> project.delete_step("normalize")  # doctest: +SKIP
    >>> project.list_steps("model_prep")  # doctest: +SKIP
    []

    Warnings
    --------
    Write operations (create_grouping, create_step, delete_step) lock the BCMS
    ruleset branch during execution and unlock it upon completion or
    failure. If an error occurs, the branch is always unlocked before
    the exception is re-raised.
    """

    CLASSIFICATION: ClassVar[str] = "analytical_project"
    """The BCMS umbrella classification this class operates on."""

    def __init__(
        self,
        client: BCMSAPIClient,
        *,
        umbrella_identifier: UUID | None = None,
        project_name: str | None = None,
    ) -> None:
        """Initialize from a BCMS umbrella.

        Exactly one of *umbrella_identifier* or *project_name* must be
        provided.

        Parameters
        ----------
        client : BCMSAPIClient
            The BCMS API client.
        umbrella_identifier : UUID | None, optional
            The identifier of the BCMS umbrella.
        project_name : str | None, optional
            The business name of the project, resolved across all domains.

        Raises
        ------
        ValueError
            If neither or both parameters are provided, if the umbrella
            does not have an 'integration' branch, if *project_name* does
            not match exactly one project, or if the umbrella is not
            classified as :attr:`CLASSIFICATION`.
        """
        self._client = client

        if umbrella_identifier is None and project_name is None:
            raise ValueError(
                "Must provide either 'umbrella_identifier' or 'project_name'."
            )
        if umbrella_identifier is not None and project_name is not None:
            raise ValueError(
                "Cannot specify both 'umbrella_identifier' and "
                "'project_name'. Provide only one."
            )

        if project_name is not None:
            umbrella_identifier = self._resolve_project_name(project_name)

        if umbrella_identifier is None:
            raise ValueError(
                "Must provide either 'umbrella_identifier' or 'project_name'."
            )
        self._umbrella_identifier: UUID = umbrella_identifier

        umbrella_data = client.query.rule_set_umbrella(identifier=umbrella_identifier)
        if umbrella_data is None:
            raise ValueError(
                f"Umbrella with identifier '{umbrella_identifier}' not found."
            )

        classification = (umbrella_data.get("classification") or {}).get("name")
        if classification not in (None, self.CLASSIFICATION):
            raise ValueError(
                f"Umbrella '{umbrella_identifier}' is classified "
                f"'{classification}'; {type(self).__name__} only operates on "
                f"'{self.CLASSIFICATION}' projects."
            )

        business_name = umbrella_data.get("businessName")
        if business_name is None:
            raise ValueError(f"Umbrella '{umbrella_identifier}' has no business name.")
        self._business_name: str = business_name
        integration_branch = None
        for branch in umbrella_data.get("branches") or []:
            if branch is not None and branch["name"] == "integration":
                integration_branch = branch
                break

        if integration_branch is None:
            raise ValueError("Umbrella does not have an 'integration' branch.")

        self._branch_identifier: UUID = integration_branch["identifier"]
        self._ruleset: RuleSet = _query_project_data(client, self._branch_identifier)
        self._ruleset_identifier: UUID = self._ruleset["identifier"]
        self._umbrella_data = umbrella_data
        self._used_components = self._query_used_components()
        self._groupings = self._extract_groupings()
        self._step_mapping = self._extract_steps()
        self._decision_table_mapping = self._extract_decision_tables()
        self._data_element_direction_mapping = (
            self._map_step_data_elements_by_direction()
        )
        self._candidates = self._extract_candidates(umbrella_data)

    def _resolve_project_name(self, project_name: str) -> UUID:
        """Resolve a business name to an umbrella identifier.

        Queries BCMS for :attr:`CLASSIFICATION` projects matching the given
        business name across all domains. Exactly one match is required.

        Parameters
        ----------
        project_name : str
            The business name of the project to find.

        Returns
        -------
        UUID
            The umbrella identifier of the matched project.

        Raises
        ------
        ValueError
            If no project matches *project_name* or if multiple
            projects match.
        """
        filter_input: RuleSetUmbrellaFilterInput = {
            "classifications": {"name": self.CLASSIFICATION},
            "businessNames": project_name,
        }

        umbrellas = self._client.query.rule_set_umbrellas(
            limit=10, offset=0, filter=filter_input
        )

        projects: list[tuple[UUID, str | None, str | None]] = []
        for umbrella_item in umbrellas["items"] or []:
            if umbrella_item is None:
                continue
            domain = umbrella_item.get("domain") or {}
            projects.append(
                (
                    umbrella_item["identifier"],
                    umbrella_item.get("alternateName"),
                    domain.get("name"),
                )
            )

        if len(projects) == 0:
            raise ValueError(f"No project found with business name '{project_name}'.")
        if len(projects) > 1:
            details = "\n".join(
                f"  - domain: {domain}, "
                f"alternate_name: {alternate_name}, "
                f"identifier: {identifier}"
                for identifier, alternate_name, domain in projects
            )
            raise ValueError(
                f"Multiple projects found with business name "
                f"'{project_name}'. An 'umbrella_identifier' must be "
                f"provided:\n{details}"
            )
        return projects[0][0]

    @property
    def dataElementTypes(self) -> dict[str, str]:
        """Map each data element name to its JSON-schema type from the contract.

        Returns
        -------
        dict[str, str]
            Dictionary mapping data element names to their JSON-schema
            types (e.g. ``"string"``, ``"number"``). Nullable types
            like ``["number", "null"]`` are resolved to the non-null type.
            Returns an empty dict if no data contract is attached.

        Examples
        --------
        >>> project.dataElementTypes  # doctest: +SKIP
        {'agre_index_id': 'string', 'pl_ptnr_mndn_cp': 'number', ...}
        """
        types: dict[str, str] = {}
        for side in ("input", "output"):
            raw_schema = self.dataContract.get(side)
            if not isinstance(raw_schema, dict):
                continue
            schema = raw_schema.get("items", raw_schema)
            if not isinstance(schema, dict):
                continue
            properties = schema.get("properties", {})
            if not isinstance(properties, dict):
                continue
            for name, spec in properties.items():
                if not isinstance(name, str) or not isinstance(spec, dict):
                    continue
                type_ = spec.get("type")
                if isinstance(type_, list):
                    type_ = next((t for t in type_ if t != "null"), None)
                if isinstance(type_, str):
                    types[name] = type_
        return types

    def _refresh_ruleset_data(self) -> None:
        """Re-query the ruleset from BCMS and rebuild derived data."""
        self._ruleset = _query_project_data(self._client, self._branch_identifier)
        self._ruleset_identifier = self._ruleset["identifier"]
        umbrella_data = self._client.query.rule_set_umbrella(
            identifier=self._umbrella_identifier
        )
        if umbrella_data is None:
            raise ValueError(
                f"Umbrella with identifier '{self._umbrella_identifier}' not found."
            )
        self._umbrella_data = umbrella_data
        self._used_components = self._query_used_components()
        self._groupings = self._extract_groupings()
        self._step_mapping = self._extract_steps()
        self._decision_table_mapping = self._extract_decision_tables()
        self._data_element_direction_mapping = (
            self._map_step_data_elements_by_direction()
        )
        self._candidates = self._extract_candidates(umbrella_data)

    def _derive_data_contract_name(self) -> str | None:
        """Derive data contract name from project name elements.

        Reads name elements from ruleset data and joins the
        ``marketable_product`` and ``family`` values with an underscore.

        Returns
        -------
        str | None
            The derived data contract name (e.g., ``"ppa_pubrec"``),
            or ``None`` if the umbrella does not have both
            ``marketable_product`` and ``family`` name elements.

        Warnings
        --------
        If the required name elements are missing, a ``UserWarning`` is
        issued explaining that the data contract cannot be auto-derived.
        """
        umbrella = self._ruleset.get("umbrella")
        name_elements = umbrella.get("nameElements") if umbrella else []

        values: dict[str, str] = {}
        for elem in name_elements or []:
            if elem is None:
                continue
            name = elem.get("name")
            value = elem.get("value")
            if name in ("marketable_product", "family") and value is not None:
                values[name] = value

        if "marketable_product" not in values or "family" not in values:
            warnings.warn(
                "Cannot derive data contract name: umbrella does not have both "
                "'marketable_product' and 'family' name elements with "
                "non-null values. Skipping data contract assignment.",
                UserWarning,
                stacklevel=2,
            )
            return None

        return f"{values['marketable_product']}_{values['family']}"

    def _resolve_data_contract(
        self,
        data_contract_name: str | None = None,
        data_contract_version: str | None = None,
    ) -> DataContractName | None:
        """Resolve a data contract by name and optional version.

        If *data_contract_name* is ``None``, the name is auto-derived
        from the project's name elements.  Queries BCMS for matching
        data contracts.  If *data_contract_version* is provided, filters
        for an exact version match; otherwise the highest version
        (by semantic versioning) is selected.

        Parameters
        ----------
        data_contract_name : str | None, optional
            The data contract name to query.  If ``None``, auto-derived
            from the umbrella's ``marketable_product`` and ``family``
            name elements.
        data_contract_version : str | None, optional
            The exact version to match.  If ``None``, the highest
            version is selected.

        Returns
        -------
        DataContractName | None
            A :class:`DataContractName` if a matching data contract is
            found, or ``None`` if no match exists.
        """
        if data_contract_name is None:
            data_contract_name = self._derive_data_contract_name()

        if data_contract_name is None:
            return None

        contracts = _query_data_contracts(self._client, data_contract_name)

        # No data contracts found matching name — return None to signal no match
        if not contracts:
            return None

        if data_contract_version is not None:
            for contract in contracts:
                if contract.get("version") == data_contract_version:
                    return DataContractName(
                        name=data_contract_name, version=data_contract_version
                    )
            return None

        # Pick highest version (semantic versioning)
        def _version_key(c: RuleSetDataContract) -> Version:
            try:
                return Version(c.get("version", "0"))
            except Exception:
                return Version("0")

        highest = max(contracts, key=_version_key)
        return DataContractName(
            name=data_contract_name, version=highest.get("version", "")
        )

    def _unlock_branch(self) -> None:
        """Unlock the integration branch.

        Safe to call even if the branch is not locked or was locked by
        a previous operation.
        """
        self._client.mutation.unlock_rule_set_branch(identifier=self._branch_identifier)

    def _ensure_branch_locked(self) -> None:
        """Ensure the branch is locked for editing.

        Queries the current branch lock status and locks it if unlocked.
        If already locked for "Editing", does nothing (idempotent).

        Raises
        ------
        PermissionError
            If branch is locked for a reason other than "Editing".
            Error message includes lock reason, responsible party,
            and timestamp.
        ValueError
            If the umbrella does not have an 'integration' branch.

        Notes
        -----
        Always call ``_unlock_branch()`` in a finally block after this
        method, regardless of whether this method acquired the lock or
        found it already locked.  This ensures proper cleanup of the
        "Editing" lock scope.

        The lock reason check only examines the ``name`` key of the
        ``lockReason`` dictionary. Any additional keys are ignored.
        """
        umbrella_data = self._client.query.rule_set_umbrella(
            identifier=self._umbrella_identifier
        )
        if umbrella_data is None:
            raise ValueError(
                f"Umbrella with identifier '{self._umbrella_identifier}' not found."
            )

        integration_branch = None
        for branch in umbrella_data.get("branches") or []:
            if branch is not None and branch["name"] == "integration":
                integration_branch = branch
                break

        if integration_branch is None:
            raise ValueError("Umbrella does not have an 'integration' branch.")

        lock_reason = integration_branch.get("lockReason")

        if lock_reason is None:
            self._client.mutation.lock_rule_set_branch(
                input=LockComponentInput(
                    identifier=self._branch_identifier,
                    lockReason=LockReason(name="Editing"),
                )
            )
            return

        lock_reason_name = (
            lock_reason.get("name") if isinstance(lock_reason, dict) else None
        )

        if lock_reason_name == "Editing":
            return

        lock_responsible = integration_branch.get("lockResponsibleParty", "unknown")
        lock_timestamp = integration_branch.get("lockTimestamp", "unknown")

        raise PermissionError(
            f"Branch is already locked. "
            f"Reason: '{lock_reason_name}', "
            f"Responsible party: {lock_responsible}, "
            f"Locked since: {lock_timestamp}"
        )

    def update_data_contract(
        self,
        data_contract_name: str | None = None,
        data_contract_version: str | None = None,
    ) -> None:
        """Update the data contract on this project.

        Resolves a data contract by name and version, then updates
        the BCMS ruleset with the resolved contract.  If no name is
        provided, it is auto-derived from the project's
        ``marketable_product`` and ``family`` name elements.

        Parameters
        ----------
        data_contract_name : str | None, optional
            The data contract name to assign.  If ``None``, the name
            is derived from the project's name elements.
        data_contract_version : str | None, optional
            The exact version to assign.  If ``None``, the highest
            available version is selected.

        Warnings
        --------
        If no matching data contract is found, a ``UserWarning`` is
        issued and no mutation is performed.

        Notes
        -----
        This method locks the BCMS ruleset branch during execution
        and unlocks it upon completion or failure.
        """
        resolved = self._resolve_data_contract(
            data_contract_name, data_contract_version
        )

        if resolved is None:
            warnings.warn(
                "No matching data contract found. The project was not updated.",
                UserWarning,
                stacklevel=2,
            )
            return

        self._ensure_branch_locked()
        try:
            self._client.mutation.update_rule_set(
                input=UpdateRuleSetInput(
                    identifier=self._ruleset_identifier,
                    dataContract=DataContractInput(
                        name=resolved.name,
                        version=resolved.version,
                    ),
                )
            )
            self._refresh_ruleset_data()
        finally:
            self._unlock_branch()

    def register_candidate(self, candidate_name: str | None = None) -> None:
        """Copy a candidate's ruleset to this project's destination umbrella.

        The destination is read from the project's destination-umbrella
        property.  When it is unset, the destination is created under
        :data:`MODEL_CLASSIFICATION` and linked back to this project.

        Parameters
        ----------
        candidate_name : str | None, optional
            Name of the candidate branch to copy (e.g. ``"candidate-1"``).
            May be omitted when the project has exactly one candidate.

        Raises
        ------
        ValueError
            If the candidate selection does not match exactly one branch.

        Notes
        -----
        Locks the destination branch during the copy and unlocks it upon
        completion or failure.
        """
        destination_id = self._find_destination()
        if destination_id is None:
            destination_id = self._create_destination()

        candidates = [
            c
            for c in self.candidates
            if candidate_name is None or c.name == candidate_name
        ]
        if len(candidates) != 1:
            named = f" named '{candidate_name}'" if candidate_name else ""
            raise ValueError(
                f"Expected exactly one candidate{named} on project "
                f"'{self.name}'; found {len(candidates)}."
            )

        source_ruleset = _query_project_data(self._client, candidates[0].identifier)
        destination = self._client.query.rule_set_umbrella(identifier=destination_id)
        destination_branch = next(
            (
                branch["identifier"]
                for branch in (destination["branches"] if destination else None) or []
                if branch and branch["name"] == "integration"
            ),
            None,
        )
        if destination_branch is None:
            raise ValueError(
                f"Destination umbrella '{destination_id}' does not have an "
                f"'integration' branch."
            )

        self._client.mutation.lock_rule_set_branch(
            input=LockComponentInput(
                identifier=destination_branch,
                lockReason=LockReason(name="Editing"),
            )
        )
        try:
            self._client.mutation.copy_rule_set_to_branch(
                input=CopyRuleSetToBranchInput(
                    sourceRuleSetIdentifier=source_ruleset["identifier"],
                    destinationBranchIdentifier=destination_branch,
                ),
                query_depth=1,
            )
        finally:
            self._client.mutation.unlock_rule_set_branch(
                identifier=destination_branch
            )

    def _find_destination(self) -> UUID | None:
        """Find the model project created from this project.

        Searches umbrellas under :data:`MODEL_CLASSIFICATION` for one whose
        :data:`SOURCE_PROPERTY` names this project.

        Returns
        -------
        UUID | None
            The identifier of the destination umbrella, or ``None`` when
            this project has no model project yet.
        """
        filter_input: RuleSetUmbrellaFilterInput = {
            "classifications": {"name": MODEL_CLASSIFICATION}
        }
        for umbrella in _query_umbrellas(self._client, filter_input):
            properties = umbrella.get("properties") or []
            for prop in properties:
                if (
                    prop is not None
                    and prop.get("name") == SOURCE_PROPERTY
                    and prop.get("value") == self.name
                ):
                    return umbrella["identifier"]
        return None

    def _create_destination(self) -> UUID:
        """Create the model project for this project.

        The new umbrella carries this project's name elements (minus the
        auto-generated version) and records this project in
        :data:`SOURCE_PROPERTY`.

        Returns
        -------
        UUID
            The identifier of the newly created umbrella.
        """
        # Imported here: project_manager imports this module at load time.
        from statefarm.bcms.analytical_projects.project_manager import ProjectManager

        domain = (self._umbrella_data.get("domain") or {}).get("name")
        name_elements = [
            {"name": elem["name"], "value": elem["value"]}
            for elem in (self._umbrella_data.get("nameElements") or [])
            if elem is not None and elem["name"] != "version"
        ]
        return ProjectManager(self._client)._create_umbrella(
            MODEL_CLASSIFICATION,
            name_elements,
            domain=domain,
            properties=[{"name": SOURCE_PROPERTY, "value": self.name}],
        )

    def package(self, branch_identifier: UUID | None = None) -> None:
        """Create a package for a project branch.

        Calls the BCMS packaging API to initiate package creation for
        the specified branch. If no branch is specified, defaults to the
        project's integration branch. The environment (test/prod) is
        automatically determined from the BCMS client URI.

        Parameters
        ----------
        branch_identifier : UUID | None, optional
            The identifier of the branch to package. If None, uses
            the project's integration branch identifier.

        Raises
        ------
        ValueError
            If the client URI does not match a supported environment.
            Only 'test' and 'prod' are supported.
        RuntimeError
            If the packaging request fails due to a non-200 response
            or a network/timeout error.

        Examples
        --------
        >>> project.package()  # doctest: +SKIP
        >>> project.package(branch_identifier="abc-123")  # doctest: +SKIP
        """
        if branch_identifier is None:
            branch_identifier = self._branch_identifier

        uri = self._client.uri
        if ".test" in uri:
            endpoint = "https://mmw.puas.test.ic1.statefarm/v1/api/package"
            api_id = "t9wnu0u5pg"
        elif ".isbx" not in uri:
            endpoint = "https://mmw.puas.ic1.statefarm/v1/api/package"
            api_id = "jgio15t0ia"
        else:
            raise ValueError(
                "Unsupported environment. Only 'test' and 'prod' "
                "environments are supported. Environment is determined "
                "from the BCMS client URI."
            )

        headers = {
            "x-apigw-api-id": api_id,
            "Authorization": f"Bearer {self._client._auth.token}",
        }
        payload = {"ruleSetBranchIdentifier": str(branch_identifier)}

        try:
            resp = requests.post(
                url=endpoint, json=payload, headers=headers, timeout=30
            )
        except Exception as e:
            raise RuntimeError(f"Packaging request failed: {e}") from e

        if resp.status_code != 200:
            raise RuntimeError(f"Packaging failed: {resp.status_code} - {resp.text}")

    def _extract_groupings(self) -> dict[str, UUID]:
        """Extract top-level groupings from ruleset data.

        Returns
        -------
        dict[str, UUID]
            Dictionary mapping grouping name to identifier.
        """
        groupings: dict[str, UUID] = {}
        for step in self._ruleset.get("steps") or []:
            if step is None:
                continue
            identifier = step.get("identifier")
            if identifier is None:
                continue
            groupings[step["name"]] = identifier
        return groupings

    def _extract_steps(self) -> dict[str, list[Step]]:
        """Extract child steps organized by grouping from ruleset data.

        Returns
        -------
        dict[str, list[Step]]
            Dictionary mapping grouping name to list of step info dicts.
            Each step dict contains all non-null fields from the BCMS
            child step data plus a 'grouping' key indicating which grouping
            it belongs to, and 'decision_tables_used' and 'data_elements_used'
            fields from used-components data.
        """
        # Build lookup from used components
        step_used_map: dict[str, object] = {}
        for suc in self._used_components.get("stepUsedComponents") or []:
            step_used_map[str(suc["identifier"])] = suc

        steps: dict[str, list[Step]] = {}
        for step in self._ruleset.get("steps") or []:
            if step is None:
                continue
            grouping_name = step["name"]
            children = step.get("children") or []
            steps[grouping_name] = []
            for child in children:
                if child is None:
                    continue
                step_info: dict[str, object] = {
                    "name": child["name"],
                    "identifier": child["identifier"],
                    "grouping": grouping_name,
                }
                for key, value in child.items():
                    if value is not None and key not in ("name", "identifier"):
                        step_info[key] = value

                # Add used-components data
                used = cast(
                    dict[str, object], step_used_map.get(str(child["identifier"]), {})
                )
                step_info["decision_tables_used"] = [
                    dt["name"]
                    for dt in cast(
                        list[dict[str, str]], used.get("decisionTablesUsed") or []
                    )
                    if dt.get("name")
                ]
                step_info["data_elements_used"] = [
                    de["name"]
                    for de in cast(
                        list[dict[str, str]], used.get("dataElementsUsed") or []
                    )
                    if de.get("name")
                ]
                steps[grouping_name].append(cast(Step, step_info))
        return steps

    def _extract_decision_tables(self) -> dict[str, DecisionTable]:
        """Extract decision tables with full data from ruleset.

        Returns
        -------
        dict[str, DecisionTable]
            Dictionary mapping decision table names to their complete
            decision table objects including identifier, data, definition,
            parameters, and other fields needed for execution.
        """
        return {dt["name"]: dt for dt in self.decisionTables}

    def _map_step_data_elements_by_direction(
        self,
    ) -> dict[str, StepDataElements]:
        """Map data elements by direction for each non-grouping step.

        Processes used components data to create a mapping of step names
        to their input and output data elements. Top-level groupings are
        excluded from this mapping.

        Returns
        -------
        dict[str, dict[str, list[str]]]
            Dictionary mapping step names to direction-organized data
            elements. Each step has a nested dict with 'input' and
            'output' keys, each containing a list of data element names.
            Both keys are always present, even if the lists are empty.
        """
        mapping: dict[str, StepDataElements] = {}

        for step_component in self._used_components.get("stepUsedComponents") or []:
            step_name = step_component.get("name")
            if not step_name or step_name in self._groupings:
                continue

            mapping[step_name] = {"input": [], "output": []}

            for data_element in step_component.get("dataElementsUsed") or []:
                elem_name = data_element.get("name")
                directions = {d.lower() for d in (data_element.get("direction") or [])}

                if not elem_name:
                    continue

                if "output" in directions:
                    mapping[step_name]["output"].append(elem_name)
                elif "input" in directions:
                    mapping[step_name]["input"].append(elem_name)

        return mapping

    def _extract_step_outputs(
        self, step: RuleStep, decision_tables: list[DecisionTable]
    ) -> set[str]:
        """Extract output data elements from a step definition using the DSL compiler.

        Parameters
        ----------
        step : RuleStep
            Step definition dict containing at least 'name' and 'executionRule'.
        decision_tables : list[DecisionTable]
            List of decision table dicts used by this step.

        Returns
        -------
        set[str]
            Set of output data element names produced by the step.
            Returns empty set if the step produces no outputs.

        Raises
        ------
        ValueError
            If the step's DSL syntax is invalid and cannot be compiled.
        """
        try:
            compiled = compile_rule_step(
                cast("dict[str, object]", step),
                decision_tables=cast(
                    "list[dict[str, object]] | None", decision_tables or None
                ),
            )
            if compiled.rule is not None and compiled.rule.outputs:
                return set(compiled.rule.outputs)
            return set()
        except Exception as e:
            raise ValueError(
                f"Failed to compile step '{step.get('name', 'unknown')}' "
                f"for output validation: {e}"
            ) from e

    def _get_all_existing_outputs(self) -> set[str]:
        """Get all output data elements from existing steps in the project.

        Returns
        -------
        set[str]
            Set of all output data element names currently produced
            by steps in the project.
        """
        all_outputs: set[str] = set()
        for directions in self._data_element_direction_mapping.values():
            all_outputs.update(directions.get("output", []))
        return all_outputs

    def _extract_candidates(self, umbrella_data: RuleSetUmbrella) -> list[Candidate]:
        """Extract candidate branches from umbrella data.

        Parameters
        ----------
        umbrella_data : RuleSetUmbrella
            The umbrella data dict as returned by the BCMS API query.

        Returns
        -------
        list[Candidate]
            List of candidate branches with ``name`` and ``identifier``.
        """
        candidates: list[Candidate] = []
        for branch in umbrella_data.get("branches") or []:
            if branch is None:
                continue
            if "candidate" in branch["name"].lower():
                candidates.append(
                    Candidate(
                        name=branch["name"],
                        identifier=branch["identifier"],
                    )
                )
        return candidates

    def _refresh_candidates(self) -> None:
        """Re-query the umbrella and update the candidates list."""
        umbrella_data = self._client.query.rule_set_umbrella(
            identifier=self._umbrella_identifier
        )
        if umbrella_data is None:
            raise ValueError(
                f"Umbrella with identifier '{self._umbrella_identifier}' not found."
            )
        self._candidates = self._extract_candidates(umbrella_data)

    def _query_used_components(self) -> RuleSetUsedComponents:
        """Query used components for the ruleset from BCMS.

        Returns
        -------
        RuleSetUsedComponents
            The used components response containing stepUsedComponents,
            ruleUsedComponents, and decisionTableUsedComponents.
        """
        return self._client.query.rule_set_used_components(
            rule_set_input=RuleSetIdentifierInput(
                ruleSetIdentifier=self._ruleset_identifier
            ),
        )

    def _insert_rule_step(self, parent: UUID | None, step: RuleStep) -> RuleStep:
        """Create a new rule step in BCMS.

        Only fields defined in ``InsertRuleStepSingle`` are passed
        through to the mutation. Positioning fields and identifiers
        are excluded since they are managed by the Project class.

        Parameters
        ----------
        parent : UUID | None
            Parent rule step identifier, or None for top-level groupings.
        step : RuleStep
            Step definition dict.

        Returns
        -------
        RuleStep
            Result from BCMS mutation.
        """
        valid_fields = set(InsertRuleStepSingle.__annotations__) - {
            "identifier",
            "parentRuleStepIdentifier",
            "afterStepIdentifier",
            "beforeStepIdentifier",
        }
        clean_step = {k: v for k, v in step.items() if k in valid_fields}
        step_input = cast(
            InsertRuleStepSingle,
            {"parentRuleStepIdentifier": parent, **clean_step},
        )

        result = self._client.mutation.insert_rule_step(
            InsertRuleStepInput(
                ruleSetIdentifier=self._ruleset_identifier,
                step=step_input,
            )
        )
        return result

    def _insert_decision_table(self, decision_table: DecisionTable) -> DecisionTable:
        """Create a new decision table in BCMS.

        Only fields defined in ``InsertDecisionTableInput`` are passed
        through to the mutation. The ``ruleSetIdentifier`` is set
        explicitly from the project context.

        Parameters
        ----------
        decision_table : DecisionTable
            Decision table definition dict.

        Returns
        -------
        DecisionTable
            Result from BCMS mutation.
        """
        valid_fields = set(InsertDecisionTableInput.__annotations__) - {
            "ruleSetIdentifier",
        }
        clean_dt = {k: v for k, v in decision_table.items() if k in valid_fields}
        dt_input = cast(
            InsertDecisionTableInput,
            {"ruleSetIdentifier": self._ruleset_identifier, **clean_dt},
        )

        result = self._client.mutation.insert_decision_table(dt_input)
        return result

    def _delete_decision_table(self, decision_table_identifier: UUID) -> None:
        """Delete a decision table in BCMS.

        Parameters
        ----------
        decision_table_identifier : UUID
            The identifier of the decision table to delete.
        """
        self._client.mutation.delete_decision_table(
            DeleteDecisionTableInput(identifier=decision_table_identifier)
        )

    def list_groupings(self) -> list[str]:
        """List all groupings in the project.

        Returns
        -------
        list[str]
            List of grouping names.
        """
        return list(self._groupings.keys())

    def create_grouping(self, grouping_name: str) -> None:
        """Create a new grouping in the project.

        Parameters
        ----------
        grouping_name : str
            Name of the grouping to create.

        Raises
        ------
        ValueError
            If a grouping with this name already exists.

        Notes
        -----
        This method locks the BCMS ruleset branch during the operation
        and unlocks it upon completion or failure.
        """
        if grouping_name in self._groupings:
            raise ValueError(f"Grouping '{grouping_name}' already exists.")

        self._ensure_branch_locked()
        try:
            step_data = {"name": grouping_name, "executionRule": None}
            self._insert_rule_step(
                parent=None,
                step=cast(RuleStep, step_data),
            )
            self._refresh_ruleset_data()
        finally:
            self._unlock_branch()

    def list_steps(self, grouping: str | list[str] | None = None) -> list[str]:
        """List step names within the project.

        Parameters
        ----------
        grouping : str | list[str] | None, optional
            Grouping name(s) to filter by. If None, returns all steps.

        Returns
        -------
        list[str]
            List of step names.

        Raises
        ------
        ValueError
            If any specified grouping does not exist.
        """
        if grouping is None:
            step_names: list[str] = []
            for children in self._step_mapping.values():
                step_names.extend(child["name"] for child in children)
            return step_names

        if isinstance(grouping, str):
            grouping = [grouping]

        invalid = [g for g in grouping if g not in self._groupings]
        if invalid:
            raise ValueError(f"Groupings do not exist: {', '.join(invalid)}")

        step_names_filtered: list[str] = []
        for g in grouping:
            step_names_filtered.extend(
                child["name"] for child in self._step_mapping.get(g, [])
            )
        return step_names_filtered

    def _validate_step_input(
        self,
        grouping: str,
        step: RuleStep,
    ) -> None:
        """Validate step input before creation.

        Parameters
        ----------
        grouping : str
            Name of the grouping to add the step to.
        step : RuleStep
            Step definition dict to validate.

        Raises
        ------
        ValueError
            If grouping does not exist, step is missing required fields,
            a step with the same name already exists, or the step produces
            output data elements that conflict with existing steps.
        """
        if grouping not in self._groupings:
            raise ValueError(f"Grouping '{grouping}' does not exist.")
        if "name" not in step:
            raise ValueError("Step dict must contain 'name'.")
        if "executionRule" not in step:
            raise ValueError("Step dict must contain 'executionRule'.")
        if step["name"] in self.list_steps():
            raise ValueError(f"Step '{step['name']}' already exists.")

        # Validate output data elements don't conflict with existing steps
        new_outputs = self._extract_step_outputs(step, [])
        if new_outputs:
            existing_outputs = self._get_all_existing_outputs()
            overlap = new_outputs & existing_outputs
            if overlap:
                raise ValueError(
                    f"Step '{step['name']}' creates output data element(s) "
                    f"that already exist in the project: "
                    f"{', '.join(sorted(overlap))}. "
                    f"Each step must produce unique output data elements."
                )

    def _validate_decision_tables(
        self,
        decision_tables: DecisionTable | list[DecisionTable] | None,
    ) -> list[DecisionTable]:
        """Normalize and validate decision tables.

        Parameters
        ----------
        decision_tables : DecisionTable | list[DecisionTable] | None
            Decision table(s) to validate. A single dict, a list of dicts,
            or None.

        Returns
        -------
        list[DecisionTable]
            Normalized list of validated decision table dicts.

        Raises
        ------
        ValueError
            If any decision table is missing required fields, has a
            duplicate name, or conflicts with existing decision tables.
        """
        if decision_tables is None:
            return []

        if isinstance(decision_tables, dict):
            dt_list = [decision_tables]
        else:
            dt_list = list(decision_tables)

        for dt in dt_list:
            if "name" not in dt:
                raise ValueError("Decision table dict must contain 'name'.")
            if "data" not in dt:
                raise ValueError("Decision table dict must contain 'data'.")
            if "definition" not in dt:
                raise ValueError("Decision table dict must contain 'definition'.")
            if dt["name"] in self._decision_table_mapping:
                raise ValueError(f"Decision table '{dt['name']}' already exists.")

        dt_names = [dt["name"] for dt in dt_list]
        if len(dt_names) != len(set(dt_names)):
            raise ValueError("Duplicate decision table names in input.")

        return dt_list

    def create_step(
        self,
        grouping: str,
        step: RuleStep,
        decision_tables: DecisionTable | list[DecisionTable] | None = None,
    ) -> list[str]:
        """Create a new step in a grouping.

        Parameters
        ----------
        grouping : str
            Name of the grouping to add the step to.
        step : RuleStep
            Step definition dict. Must contain 'name' and 'executionRule'.
        decision_tables : DecisionTable | list[DecisionTable] | None, optional
            Decision table(s) to associate with the step.

        Returns
        -------
        list[str]
            List containing the created step name.

        Raises
        ------
        ValueError
            If validation fails for grouping, step, decision tables,
            or if the step produces output data elements that conflict
            with existing steps in the project.

        Notes
        -----
        This method locks the BCMS ruleset branch during the operation
        and unlocks it upon completion or failure. If the step is created
        but a decision table creation fails, a best-effort rollback of the
        step is attempted before re-raising the exception.
        """
        dt_list = self._validate_decision_tables(decision_tables)
        self._validate_step_input(grouping, step)

        self._ensure_branch_locked()

        created_step_identifier: UUID | None = None
        try:
            parent_identifier = self._groupings[grouping]

            step_result = self._insert_rule_step(
                parent=parent_identifier,
                step=step,
            )
            created_step_identifier = (
                step_result["identifier"] if step_result.get("identifier") else None
            )

            for dt in dt_list:
                self._insert_decision_table(
                    decision_table=dt,
                )

            self._refresh_ruleset_data()
        except Exception:
            if created_step_identifier is not None:
                try:
                    self._client.mutation.delete_rule_step(
                        DeleteRuleStepInput(identifier=created_step_identifier)
                    )
                except Exception:  # noqa: S110
                    pass
            raise
        finally:
            self._unlock_branch()

        return [step["name"]]

    def _resolve_step_names(
        self,
        step_names: str | list[str] | None,
        groupings: str | list[str] | None,
    ) -> list[str] | None:
        """Resolve and validate step names from get_steps parameters.

        Parameters
        ----------
        step_names : str | list[str] | None
            Step name(s) provided directly.
        groupings : str | list[str] | None
            Grouping name(s) to retrieve all steps from.

        Returns
        -------
        list[str] | None
            Resolved list of step names, or None if no steps match.
            When both parameters are None, returns all non-grouping
            step names.

        Raises
        ------
        ValueError
            If both parameters are provided, or groupings
            does not exist.
        """
        # Treat empty list as "not provided"
        if isinstance(groupings, list) and len(groupings) == 0:
            groupings = None

        if step_names is not None and groupings is not None:
            raise ValueError(
                "Cannot specify both 'step_names' and 'groupings'. Provide only one."
            )

        # Neither provided — return all non-grouping steps
        if step_names is None and groupings is None:
            all_names: list[str] = []
            for children in self._step_mapping.values():
                all_names.extend(child["name"] for child in children)
            return all_names if all_names else None

        if groupings is not None:
            if isinstance(groupings, str):
                groupings = [groupings]

            invalid = [g for g in groupings if g not in self._groupings]
            if invalid:
                raise ValueError(f"Groupings do not exist: {', '.join(invalid)}")

            resolved: list[str] = []
            for g in groupings:
                resolved.extend(step["name"] for step in self._step_mapping.get(g, []))
            return resolved if resolved else None

        if isinstance(step_names, str):
            return [step_names]
        return list(step_names or [])

    def get_steps(
        self,
        step_names: str | list[str] | None = None,
        groupings: str | list[str] | None = None,
    ) -> Steps:
        """Get step data and associated decision tables.

        Retrieve steps by name, by grouping, or all non-grouping steps.
        At most one of ``step_names`` or ``groupings`` may be provided.
        If neither is provided, all non-grouping steps are returned.

        Parameters
        ----------
        step_names : str | list[str] | None, optional
            Name or list of names of steps to retrieve.
        groupings : str | list[str] | None, optional
            Name or list of names of groupings to retrieve all steps
            from. When a list is provided, steps from all specified
            groupings are returned combined.

        Returns
        -------
        Steps
            Dictionary with 'steps' and 'decisionTables' keys.
            'steps' is a list of step data dicts (each including
            'decision_tables_used' and 'data_elements_used' fields).
            'decisionTables' is a list of full decision table objects
            used by any of the requested steps (deduplicated).

        Raises
        ------
        ValueError
            If both ``step_names`` and ``groupings`` are provided,
            if a step does not exist, or if the grouping does not
            exist.

        Examples
        --------
        >>> result = project.get_steps()  # all non-grouping steps  # doctest: +SKIP
        >>> result = project.get_steps("normalize")  # doctest: +SKIP
        >>> result["steps"][0]["name"]  # doctest: +SKIP
        'normalize'
        >>> result["decisionTables"]  # doctest: +SKIP
        [...]
        >>> result = project.get_steps(["normalize", "score"])  # doctest: +SKIP
        >>> result = project.get_steps(groupings="model_prep")  # doctest: +SKIP
        >>> len(result["steps"])  # all steps in model_prep  # doctest: +SKIP
        6
        >>> result = project.get_steps(groupings=["model_prep", "model"])  # doctest: +SKIP
        """
        resolved_names = self._resolve_step_names(step_names, groupings)
        if resolved_names is None:
            return cast(Steps, {"steps": [], "decisionTables": []})

        steps_result: list[Step] = []
        dt_names_seen: set[str] = set()

        for name in resolved_names:
            step_data = self._get_step_data_by_name(name)
            steps_result.append(step_data)
            for dt_name in step_data.get("decision_tables_used", []):
                dt_names_seen.add(dt_name)

        # Collect full decision table objects
        decision_tables_result: list[DecisionTable] = []
        for dt_name in sorted(dt_names_seen):
            if dt_name in self._decision_table_mapping:
                decision_tables_result.append(self._decision_table_mapping[dt_name])

        return cast(
            Steps, {"steps": steps_result, "decisionTables": decision_tables_result}
        )

    def _get_step_data_by_name(self, step_name: str) -> Step:
        """Look up a step by name from cached steps data.

        Parameters
        ----------
        step_name : str
            Name of the step to look up.

        Returns
        -------
        Step
            The step data dictionary.

        Raises
        ------
        ValueError
            If the step does not exist.
        """
        for children in self._step_mapping.values():
            for child in children:
                if child["name"] == step_name:
                    return child

        raise ValueError(f"Step '{step_name}' does not exist.")

    def _find_exclusive_decision_tables_by_name(
        self, step_name: str, dt_names: list[str]
    ) -> list[UUID]:
        """Identify decision tables used exclusively by one step.

        Uses cached used-components data to determine exclusivity
        without additional API calls.

        Parameters
        ----------
        step_name : str
            The name of the step being deleted.
        dt_names : list[str]
            List of decision table names used by the step.

        Returns
        -------
        list[UUID]
            Identifiers of decision tables used only by *step_name*.
        """
        # Build set of all DT names used by OTHER steps
        other_steps_dt_names: set[str] = set()
        for children in self._step_mapping.values():
            for child in children:
                if child["name"] != step_name:
                    other_steps_dt_names.update(child.get("decision_tables_used", []))

        exclusive: list[UUID] = []
        for dt_name in dt_names:
            if dt_name not in other_steps_dt_names:
                dt_obj = self._decision_table_mapping.get(dt_name)
                if dt_obj and dt_obj.get("identifier"):
                    exclusive.append(dt_obj["identifier"])

        return exclusive

    def _delete_decision_tables_with_retry(self, dt_identifiers: list[UUID]) -> None:
        """Delete decision tables with retry logic.

        Parameters
        ----------
        dt_identifiers : list[UUID]
            Identifiers of the decision tables to delete.

        Warnings
        --------
        Issues a ``UserWarning`` for each decision table that cannot be
        deleted after 3 attempts.
        """
        for dt_identifier in dt_identifiers:
            dt_deleted = False
            for attempt in range(3):
                try:
                    self._delete_decision_table(dt_identifier)
                    dt_deleted = True
                    break
                except Exception:
                    if attempt < 2:
                        time.sleep(0.1)

            if not dt_deleted:
                warnings.warn(
                    f"Failed to delete decision table '{dt_identifier}' "
                    f"after 3 attempts. The step has been deleted, but "
                    f"this decision table remains.",
                    UserWarning,
                    stacklevel=3,
                )

    def delete_step(self, step_name: str) -> None:
        """Delete a step and its exclusively used decision tables.

        Removes a step from the project and deletes any decision tables
        that are used exclusively by this step. Decision tables shared
        with other steps are preserved.

        Parameters
        ----------
        step_name : str
            Name of the step to delete.

        Raises
        ------
        ValueError
            If the step does not exist.
        RuntimeError
            If step deletion fails after 3 retry attempts.
        PermissionError
            If branch is locked for a reason other than "Editing".

        Warnings
        --------
        If a decision table deletion fails after 3 retries, a
        ``UserWarning`` is issued but the operation continues.  The
        step is still deleted even if some decision tables cannot be
        removed.

        Notes
        -----
        This method locks the BCMS ruleset branch during the operation
        and unlocks it upon completion or failure.

        Decision tables are considered "exclusively used" if they are
        only referenced by the step being deleted.  Exclusivity is
        determined from cached used-components data.

        Examples
        --------
        >>> project.delete_step("normalize")  # doctest: +SKIP
        >>> project.list_steps("model_prep")  # doctest: +SKIP
        []
        """
        step_data = self._get_step_data_by_name(step_name)
        step_identifier = step_data["identifier"]
        dt_names = step_data.get("decision_tables_used", [])

        # Determine which decision tables are exclusively used by this step
        exclusive_dt_identifiers = self._find_exclusive_decision_tables_by_name(
            step_name, dt_names
        )

        # Lock branch
        self._ensure_branch_locked()

        try:
            # Delete step with retry logic
            for attempt in range(3):
                try:
                    self._client.mutation.delete_rule_step(
                        DeleteRuleStepInput(identifier=step_identifier)
                    )
                    break
                except Exception as err:
                    if attempt < 2:
                        time.sleep(0.1)
                    else:
                        raise RuntimeError(
                            f"Failed to delete step '{step_name}' after 3 attempts."
                        ) from err

            # Delete exclusive decision tables with retry logic
            self._delete_decision_tables_with_retry(exclusive_dt_identifiers)

            self._refresh_ruleset_data()
        finally:
            self._unlock_branch()

    def _build_producer_index(self) -> dict[str, list[str]]:
        """Build reverse index mapping data elements to producing steps.

        Returns
        -------
        dict[str, list[str]]
            Mapping of data element names to lists of step names that
            produce them.
        """
        producers: dict[str, list[str]] = {}
        for step_name, directions in self._data_element_direction_mapping.items():
            for output_elem in directions["output"]:
                if output_elem not in producers:
                    producers[output_elem] = []
                producers[output_elem].append(step_name)
        return producers

    def get_step_dependencies(self, step_names: str | list[str]) -> list[str]:
        """Get all steps required to execute the specified steps.

        Traverses the data element dependency graph to find all upstream
        steps that produce data elements required by the specified steps.
        The result includes the original steps and all their transitive
        dependencies.

        Parameters
        ----------
        step_names : str | list[str]
            Name or list of names of steps to find dependencies for.

        Returns
        -------
        list[str]
            List of all step names including the requested steps and
            all steps they depend on. Steps are returned in no
            particular order.

        Raises
        ------
        ValueError
            If any of the specified step names do not exist in the
            project.

        Examples
        --------
        >>> deps = project.get_step_dependencies("busn_model_snum")  # doctest: +SKIP
        >>> "model_scaled_score" in deps  # doctest: +SKIP
        True

        >>> deps = project.get_step_dependencies(["step1", "step2"])  # doctest: +SKIP

        Notes
        -----
        - External data elements (not produced by any step in the
          project) are silently ignored during traversal.
        - If multiple steps produce the same data element, all
          producers are included in the result.
        - The method handles cases where steps reference each other's
          outputs without entering infinite loops.
        """
        if isinstance(step_names, str):
            step_names = [step_names]

        if not step_names:
            return []

        for name in step_names:
            if name not in self._data_element_direction_mapping:
                raise ValueError(f"Step '{name}' does not exist.")

        producers = self._build_producer_index()

        # Traverse dependency graph breadth-first
        all_steps: set[str] = set(step_names)
        to_process: list[str] = list(step_names)

        while to_process:
            current_step = to_process.pop(0)
            step_info = self._data_element_direction_mapping.get(current_step, {})
            required_inputs = step_info.get("input", [])

            for input_elem in required_inputs:
                for producer in producers.get(input_elem, []):
                    if producer not in all_steps:
                        all_steps.add(producer)
                        to_process.append(producer)

        return list(all_steps)

    def _next_candidate_number(self) -> int:
        """Determine the next candidate branch number.

        Examines existing candidate names to find the highest number
        and returns the next sequential number.

        Returns
        -------
        int
            The next candidate number to use.
        """
        existing_numbers: list[int] = []
        for candidate in self._candidates:
            match = re.search(r"candidate-(\d+)", candidate.name, re.IGNORECASE)
            if match:
                existing_numbers.append(int(match.group(1)))
        return max(existing_numbers) + 1 if existing_numbers else 1

    def _build_pruned_ruleset(
        self,
        required_steps: set[str],
    ) -> tuple[list[RuleStep], list[DecisionTable]]:
        """Build pruned steps and decision tables containing only required steps.

        Filters the integration branch's ruleset to include only the specified
        steps and their associated decision tables. Preserves grouping structure
        and step ordering, removing groupings that would be empty. Identifiers
        are stripped from the output since the candidate branch has its own
        identifiers assigned by BCMS.

        Parameters
        ----------
        required_steps : set[str]
            Set of step names to retain in the pruned ruleset.

        Returns
        -------
        tuple[list[RuleStep], list[DecisionTable]]
            A tuple of (pruned_steps, pruned_decision_tables) ready for
            use with UpdateRuleSetInput.
        """
        # Build pruned steps - filter children within each grouping
        pruned_steps: list[RuleStep] = []
        for grouping in self.steps or []:
            children = grouping.get("children") or []
            pruned_children = [
                {k: v for k, v in child.items() if k != "identifier" and v is not None}
                for child in children
                if child["name"] in required_steps
            ]
            if pruned_children:
                pruned_grouping: dict[str, object] = {
                    k: v
                    for k, v in grouping.items()
                    if k not in ("children", "identifier") and v is not None
                }
                pruned_grouping["children"] = pruned_children
                pruned_steps.append(cast(RuleStep, pruned_grouping))

        # Identify required DT names from stepMapping
        required_dt_names: set[str] = set()
        for steps_list in self.stepMapping.values():
            for step in steps_list:
                if step["name"] in required_steps:
                    required_dt_names.update(step["decision_tables_used"])

        # Filter decision tables using decisionTableMapping for O(1) lookups
        dt_map = self.decisionTableMapping
        pruned_dts: list[DecisionTable] = cast(
            list[DecisionTable],
            [
                {
                    k: v
                    for k, v in dt_map[dt_name].items()
                    if k != "identifier" and v is not None
                }
                for dt_name in required_dt_names
                if dt_name in dt_map
            ],
        )

        return pruned_steps, pruned_dts

    def create_candidate(
        self,
        step_names: str | list[str],
        comment: str | None = None,
    ) -> Candidate:
        """Create a candidate branch containing only the specified steps and dependencies.

        Creates a new candidate branch as a copy of integration, then replaces
        the ruleset content with a pruned version containing only the required
        steps and their decision tables.

        Parameters
        ----------
        step_names : str | list[str]
            Name or list of names of steps to include in the candidate.
            All transitive dependencies are automatically included.
        comment : str | None, optional
            An optional comment describing the candidate branch.

        Returns
        -------
        Candidate
            The created candidate branch, with ``name`` and ``identifier``.

        Raises
        ------
        ValueError
            If any of the specified step names do not exist in the project.
        RuntimeError
            If the candidate creation or pruning fails. A warning is
            issued if the branch could not be cleaned up after failure.

        Examples
        --------
        >>> result = project.create_candidate("busn_model_snum")  # doctest: +SKIP
        >>> result.name  # doctest: +SKIP
        'candidate-1'

        >>> result = project.create_candidate(  # doctest: +SKIP
        ...     ["step_a", "step_b"],
        ...     comment="Release candidate for Q1",
        ... )
        """
        if isinstance(step_names, str):
            step_names = [step_names]

        # Get all required steps (validates step names exist)
        required_steps = set(self.get_step_dependencies(step_names))

        # Build pruned ruleset from integration branch data
        pruned_steps, pruned_dts = self._build_pruned_ruleset(required_steps)

        # Determine next candidate number
        candidate_name = f"candidate-{self._next_candidate_number()}"

        # Create branch as copy of integration
        branch_result = self._client.mutation.insert_rule_set_branch(
            InsertRuleSetBranchInput(
                ruleSetUmbrellaIdentifier=self._umbrella_identifier,
                name=candidate_name,
                comment=comment,
            )
        )
        candidate_branch_id: UUID = branch_result["identifier"]

        try:
            # Query candidate's ruleset to get its identifier
            candidate_ruleset = _query_project_data(self._client, candidate_branch_id)
            candidate_ruleset_id: UUID = candidate_ruleset["identifier"]

            # Lock candidate branch, update ruleset, unlock
            self._client.mutation.lock_rule_set_branch(
                input=LockComponentInput(
                    identifier=candidate_branch_id,
                    lockReason=LockReason(name="Editing"),
                )
            )
            try:
                update_input = cast(
                    UpdateRuleSetInput,
                    {
                        "identifier": candidate_ruleset_id,
                        "steps": pruned_steps,
                        "decisionTables": pruned_dts,
                    },
                )
                self._client.mutation.update_rule_set(input=update_input)
            finally:
                self._client.mutation.unlock_rule_set_branch(
                    identifier=candidate_branch_id
                )

            # Package the candidate
            self.package(branch_identifier=candidate_branch_id)

            # Refresh candidates list
            self._refresh_candidates()

        except Exception as e:
            warnings.warn(
                f"Candidate branch '{candidate_name}' (identifier: "
                f"{candidate_branch_id}) may have been partially created. "
                f"Manual cleanup may be required. Error: {e}",
                UserWarning,
                stacklevel=2,
            )
            raise

        return Candidate(name=candidate_name, identifier=candidate_branch_id)

    def __str__(self) -> str:
        """Return string representation of the project."""
        return f"{type(self).__name__}(umbrella_identifier={self._umbrella_identifier})"

    def __repr__(self) -> str:
        """Return string representation of the project."""
        return self.__str__()

"""Standard Queries for Analytical Projects."""

from __future__ import annotations

from uuid import UUID

from statefarm.bcontent.sdk.api import BCMSAPIClient
from statefarm.bcontent.sdk.api.bcms import (
    RuleSet,
    RuleSetDataContract,
    RuleSetDataContractFilterInput,
    RuleSetIdentifierInput,
    RuleSetUmbrella,
    RuleSetUmbrellaFilterInput,
)


def _query_umbrellas(
    client: BCMSAPIClient,
    filter_input: RuleSetUmbrellaFilterInput,
    limit: int = 50,
) -> list[RuleSetUmbrella]:
    """Query rule set umbrellas from BCMS, following pagination.

    Parameters
    ----------
    client : BCMSAPIClient
        The BCMS client instance.
    filter_input : RuleSetUmbrellaFilterInput
        The umbrella filter to query with.
    limit : int, optional
        Maximum number of items per page (default: 50).

    Returns
    -------
    list[RuleSetUmbrella]
        All matching umbrellas across every page.
    """
    umbrellas: list[RuleSetUmbrella] = []
    offset = 0
    while True:
        result = client.query.rule_set_umbrellas(
            limit=limit, offset=offset, filter=filter_input
        )
        umbrellas.extend(u for u in (result["items"] or []) if u is not None)
        if result["nextOffset"] is None:
            break
        offset = result["nextOffset"]
    return umbrellas


def _query_project_data(client: BCMSAPIClient, branch_identifier: UUID) -> RuleSet:
    """Query project data from BCMS."""
    project_info = client.query.rule_set(
        rule_set_input=RuleSetIdentifierInput(ruleSetBranchIdentifier=branch_identifier)
    )
    if project_info is None:
        raise ValueError(
            f"Rule set not found for branch identifier '{branch_identifier}'."
        )
    return project_info


def _query_data_contracts(
    client: BCMSAPIClient,
    name: str,
    limit: int = 50,
) -> list[RuleSetDataContract]:
    """Query data contracts from BCMS by name.

    Parameters
    ----------
    client : BCMSAPIClient
        The BCMS client instance.
    name : str
        The data contract name to filter by.
    limit : int, optional
        Maximum number of items per page (default: 50).

    Returns
    -------
    list[RuleSetDataContract]
        List of data contract items. Each item contains keys such as
        ``identifier``, ``name``, ``version``, ``input``, and ``output``.
    """
    filter_input = RuleSetDataContractFilterInput(names=[name])
    contracts: list[RuleSetDataContract] = []
    offset = 0
    while True:
        result = client.query.rule_set_data_contracts(
            limit=limit, offset=offset, filter=filter_input
        )
        contracts.extend(result["items"] or [])
        if result["nextOffset"] is None:
            break
        offset = result["nextOffset"]
    return contracts

def test_create_grouping_success(bcms_client, mock_umbrella_data, mock_ruleset_data):
    bcms_client.query.rule_set_umbrella.return_value = mock_umbrella_data
    bcms_client.query.rule_set.return_value = mock_ruleset_data
    proj = Project(bcms_client, umbrella_identifier="test-umbrella-id")

    # After creation, refresh will return updated data
    updated_ruleset = copy.deepcopy(mock_ruleset_data)
    updated_ruleset["steps"].append(
        {
            "name": "new_grouping",
            "identifier": "new-grouping-id",
            "level": 0,
            "children": [],
        }
    )

    # First call returns original, second call (refresh) returns updated
    bcms_client.query.rule_set.side_effect = [
        mock_ruleset_data,
        updated_ruleset,
    ]
    # Re-create to get fresh side_effect
    proj = Project(bcms_client, umbrella_identifier="test-umbrella-id")

    bcms_client.query.rule_set.side_effect = [updated_ruleset]
    proj.create_grouping("new_grouping")

    assert bcms_client.mutation.lock_rule_set_branch.call_count == 1
    assert bcms_client.mutation.unlock_rule_set_branch.call_count == 1
    assert "new_grouping" in proj.list_groupings()

class TestRegisterCandidate:

    def _make_project(self, bcms_client, umbrella, ruleset, linked=True):
        bcms_client.query.rule_set_umbrella.return_value = umbrella
        bcms_client.query.rule_set.return_value = ruleset
        bcms_client.query.rule_set_umbrellas.return_value = {
            "items": [
                {
                    "identifier": "dest-umb",
                    "properties": [
                        {"name": "analytical_project_source", "value": "bn"}
                    ],
                }
            ]
            if linked
            else [],
            "nextOffset": None,
        }
        return Project(bcms_client, umbrella_identifier="umb")

    @pytest.fixture
    def umbrella(self):
        return {
            "identifier": "umb",
            "businessName": "bn",
            "alternateName": "alt",
            "domain": {"name": "auto_rating"},
            "nameElements": [{"name": "family", "value": "credit"}],
            "branches": [
                {"name": "integration", "identifier": "br", "lockReason": None},
                {"name": "candidate-1", "identifier": "cand-br", "lockReason": None},
            ],
        }

    def ruleset(self):
        return {"identifier": "rs", "steps": [], "decisionTables": []}

    def test_copies_candidate_to_destination(self, bcms_client, umbrella, ruleset):

        bcms_client.query.rule_set_umbrella.return_value = {
            "identifier": "dest-umb",
            "branches": [{"name": "integration", "identifier": "dest-br"}],

            "input"
